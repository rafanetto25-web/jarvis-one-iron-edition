@JavascriptInterface public void stopVoiceService() { runOnUiThread(() -> { Intent service = new Intent( MainActivity.this, JarvisVoiceService.class ); try { stopService(service); Toast.makeText( MainActivity.this, "Escuta do JARVIS interrompida.", Toast.LENGTH_SHORT ).show(); } catch (Exception e) { Toast.makeText( MainActivity.this, "Não foi possível parar a escuta.", Toast.LENGTH_SHORT ).show(); } }); }
